Lyrics (multi-line)

To accommodate lyrics with multiple lines (i.e., multiple verses), lyric lines are encoded as structured objects. Each event's "lyrics" object has a "lines" array, with each line representing a vertical column within that syllable. Each line is identified by an ID string, which is user-defined within the document. By default, lyric lines are ordered alphabetically by the ID string.

{ "mnx": { "version": 1 }, "global": { "lyrics": { "lineOrder": [ "1", "2" ] }, "measures": [ { "time": { "count": 4, "unit": 4 } } ] }, "parts": [ { "measures": [ { "clefs": [ { "clef": { "sign": "G", "staffPosition": -2 } } ], "sequences": [ { "content": [ { "duration": { "base": "quarter" }, "lyrics": { "lines": { "1": { "text": "Are" }, "2": { "text": "Am" } } }, "notes": [ { "pitch": { "octave": 5, "step": "C" } } ] }, { "duration": { "base": "quarter" }, "lyrics": { "lines": { "1": { "text": "you" }, "2": { "text": "I" } } }, "notes": [ { "pitch": { "octave": 5, "step": "D" } } ] }, { "duration": { "base": "quarter" }, "lyrics": { "lines": { "1": { "type": "start", "text": "sleep" }, "2": { "type": "start", "text": "sleep" } } }, "notes": [ { "pitch": { "octave": 5, "step": "E" } } ] }, { "duration": { "base": "quarter" }, "lyrics": { "lines": { "1": { "type": "end", "text": "ing?" }, "2": { "type": "end", "text": "ing?" } } }, "notes": [ { "pitch": { "octave": 5, "step": "C" } } ] } ] } ] } ] } ] }
